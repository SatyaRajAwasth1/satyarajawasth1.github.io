\section{Research and Academic Projects}
\project{sysCrow: Log-Based Anomaly Detection and Fault Localization}
        {Minor project, B.E.\ in Software Engineering, NCIT}
        {\href{https://github.com/SatyaRajAwasth1/sysCrow}{github.com/SatyaRajAwasth1/sysCrow}}
\begin{itemize}
  \item Built an end-to-end system that detects anomalous behavior in system logs and points to the
        components and events most likely responsible for a failure.
  \item Implemented sequence-based anomaly detection with LSTM models, inspired by DeepLog (Du et al.,
        ACM CCS 2017), and fault localization through unsupervised log segmentation, adapted from
        Dobrowolski et al.\ (Applied Sciences, 2024). \textit{Python, LSTM.}
\end{itemize}
\gap
\project{SwiftCache: A High-Performance In-Memory Caching Database}
        {Final-year major project, B.E.\ in Software Engineering, NCIT}{}
\begin{itemize}
  \item Designed and implemented an in-memory caching database focused on low latency, high
        throughput, memory efficiency, and concurrent data structures.
  \item Implemented LRU and S3-FIFO eviction alongside a lightweight ML-based eviction policy; explored
        forkless snapshotting and Zstd compression for persistence, and benchmarked performance under
        high load. \textit{Rust, Tokio, Zstd.}
\end{itemize}
\gap
\project{Secure Data Migration System over TCP}
        {Network Programming coursework, NCIT}
        {\href{https://github.com/SatyaRajAwasth1/tcp-data-migration-system}{github.com/SatyaRajAwasth1/tcp-data-migration-system}}
\begin{itemize}
  \item Designed a custom application-layer protocol over TCP that migrates directories by transferring
        only missing or modified files, detected by comparing SHA-256 hashes.
  \item Implemented chunked streaming, resumable transfers from the last byte received, end-to-end
        integrity verification, and multiplexed connections with \texttt{select()}.
        \textit{C, POSIX sockets.}
\end{itemize}
